## 图2 综合案例三法分析（发现不达标）

### 仿真代码

```python
"""第9章图2：G=6/[s(s+1)(s+4)] 的三法分析"""
import numpy as np
import matplotlib.pyplot as plt
import control

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

s = control.TransferFunction.s
G = 6/(s*(s+1)*(s+4))
sys_cl = control.feedback(G, 1)

fig = plt.figure(figsize=(14, 4.5))
# ① 根轨迹
ax1 = fig.add_subplot(1, 3, 1)
control.root_locus(G, kvect=np.linspace(0, 40, 600), plot=True, ax=ax1)
ax1.set_xlim(-5, 1.5); ax1.set_ylim(-3.5, 3.5)
ax1.axhline(0, color='k', lw=0.8); ax1.axvline(0, color='k', lw=0.8)
for p in [0, -1, -4]:
    ax1.plot(p, 0, 'x', ms=12, mew=3, color='#C62828')
ax1.plot(0, 2, 'o', ms=8, color='#E65100'); ax1.plot(0, -2, 'o', ms=8, color='#E65100')
ax1.annotate('临界点 s=±j2\nK=20', xy=(0, 2), xytext=(0.6, 2.7),
             arrowprops=dict(arrowstyle='->', color='#E65100'), color='#E65100', fontsize=9)
ax1.set_title('① 根轨迹：极点随 K 走位\nK=6 时在左半平面（稳定）', fontsize=10)
ax1.grid(alpha=0.3)

# ② Bode 图与裕度
gm, pm, wg, wc = control.margin(G)
omega = np.logspace(-2, 1.5, 600)
mag, phase, _ = control.frequency_response(G, omega)
ax2 = fig.add_subplot(2, 3, 2)
ax2.semilogx(omega, 20*np.log10(np.maximum(mag, 1e-9)), lw=2, color='#1565C0')
ax2.axhline(0, color='k', lw=0.8)
ax2.axvline(wc, color='#E65100', ls='--', lw=1.2)
ax2.text(wc, 5, f'ωc={wc:.2f}', color='#E65100', fontsize=9)
ax2.text(0.012, -25, f'γ={pm:.1f}°', fontsize=11, color='#C62828')
ax2.set_title('② Bode 幅频：γ=%d° < 45°\n不达标 → 需要校正' % pm, fontsize=10)
ax2.set_xlabel('ω'); ax2.set_ylabel('dB'); ax2.grid(alpha=0.3, which='both')

ax3 = fig.add_subplot(2, 3, 5)
ax3.semilogx(omega, np.degrees(phase), lw=2, color='#C62828')
ax3.axhline(-180, color='k', lw=0.8, ls=':')
ax3.axvline(wc, color='#E65100', ls='--', lw=1.2)
ax3.set_xlabel('ω'); ax3.set_ylabel('相角°'); ax3.grid(alpha=0.3, which='both')

# ③ 时域响应
ax4 = fig.add_subplot(1, 3, 3)
t = np.linspace(0, 20, 1000)
T, y = control.step_response(sys_cl, t)
ax4.plot(T, y, lw=2.2, color='#2E7D32')
ax4.axhline(1, color='k', ls=':', lw=1)
ax4.set_xlim(0, 20); ax4.set_ylim(-0.3, 1.8)
ax4.set_xlabel('t (s)'); ax4.set_ylabel('c(t)')
ax4.set_title('③ 时域响应：超调大、振荡久\nσ%≈50% 不达标', fontsize=10)
ax4.grid(alpha=0.3)

fig.suptitle('综合案例 G=6/[s(s+1)(s+4)]：三法齐上阵，结论一致——稳定但不达标', fontsize=13)
fig.tight_layout(rect=[0, 0, 1, 0.93])
fig.savefig('figures/02_综合案例三法分析.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图2 已生成，γ=%.1f°, ωc=%.2f' % (pm, wc))
```

### 运行与结果

![综合案例三法分析](./figures/02_综合案例三法分析.png)

*图2 左：根轨迹显示 K=6 时闭环极点全部在左半平面（稳定）；中：Bode 图读出 γ≈30°<45°（不达标）；右：阶跃响应超调约 50%、振荡持久（不达标）。三法结论一致：稳定，但"脆"且"慢"。*

**现象解释**：这就是完整的"分析"阶段——
- **稳**：三种方法都确认系统稳定（Routh 全正 / 根轨迹不过虚轴 / γ>0）；
- **快、稳**：γ≈30° 太小、σ%≈50% 太大 → **不达标**；
- **准**：Ⅰ型系统速度输入误差 $e_{ss}=1/6$（可由 K 直接算出）。

**工程结论**：稳定只是底线，性能不达标就得改——进入校正。

## 图3 校正前后三法验证（达标）

### 仿真代码

```python
"""第9章图3：超前校正 + 三法验证"""
import numpy as np
import matplotlib.pyplot as plt
import control

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

s = control.TransferFunction.s
G = 6/(s*(s+1)*(s+4))
sys_cl = control.feedback(G, 1)
gm, pm, wg, wc = control.margin(G)

# 超前校正：φm = 45-pm+12，a = (1+sinφm)/(1-sinφm)
phi_m = np.radians(45-pm+12)
a = (1+np.sin(phi_m))/(1-np.sin(phi_m))
omega_s = np.logspace(-2, 1.5, 4000)
mag0, _, _ = control.frequency_response(G, omega_s)
idx = np.argmin(np.abs(20*np.log10(np.maximum(mag0, 1e-9)) + 10*np.log10(a)))
wm = omega_s[idx]
T_lead = 1/(wm*np.sqrt(a))
Gc = (1+a*T_lead*s)/(1+T_lead*s)

sys1 = control.feedback(Gc*G, 1)
gm1, pm1, _, wc1 = control.margin(Gc*G)
omega = np.logspace(-2, 1.5, 600)
mag0b, _, _ = control.frequency_response(G, omega)
mag1b, _, _ = control.frequency_response(Gc*G, omega)

fig, axes = plt.subplots(1, 3, figsize=(14, 4.5))
t = np.linspace(0, 20, 1000)
T, y0 = control.step_response(sys_cl, t)
T, y1 = control.step_response(sys1, t)
axes[0].plot(T, y0, lw=2.2, color='#C62828', label='校正前 γ=%.0f°' % pm)
axes[0].plot(T, y1, lw=2.2, color='#2E7D32', label='校正后 γ=%.0f°' % pm1)
axes[0].axhline(1, color='k', ls=':', lw=1)
axes[0].set_xlim(0, 20); axes[0].set_ylim(-0.3, 1.8)
axes[0].set_xlabel('t (s)'); axes[0].set_ylabel('c(t)')
axes[0].set_title('时域验证：超调下降、响应加快', fontsize=10)
axes[0].legend(fontsize=9); axes[0].grid(alpha=0.3)

control.root_locus(Gc*G, kvect=np.linspace(0, 40, 600), plot=True, ax=axes[1])
axes[1].set_xlim(-5, 1.5); axes[1].set_ylim(-3.5, 3.5)
axes[1].axhline(0, color='k', lw=0.8); axes[1].axvline(0, color='k', lw=0.8)
axes[1].set_title('根轨迹验证：极点左移、更稳', fontsize=10)
axes[1].grid(alpha=0.3)

axes[2].semilogx(omega, 20*np.log10(np.maximum(mag0b, 1e-9)), lw=2, color='#C62828', label='校正前')
axes[2].semilogx(omega, 20*np.log10(np.maximum(mag1b, 1e-9)), lw=2, color='#2E7D32', label='校正后')
axes[2].axhline(0, color='k', lw=0.8)
axes[2].axvline(wc, color='#E65100', ls='--', lw=1.2)
axes[2].axvline(wc1, color='#2E7D32', ls='--', lw=1.2)
axes[2].text(0.012, -25, f'γ: {pm:.0f}°→{pm1:.0f}°', fontsize=11, color='#333')
axes[2].set_xlabel('ω'); axes[2].set_ylabel('dB')
axes[2].set_title('频域验证：ωc 右移、γ 达标', fontsize=10)
axes[2].legend(fontsize=9, loc='lower left'); axes[2].grid(alpha=0.3, which='both')

fig.suptitle('校正前后三法验证：结论一致——达标！(γ: %d° → %d°, a=%.2f)' % (pm, pm1, a), fontsize=13)
fig.tight_layout(rect=[0, 0, 1, 0.93])
fig.savefig('figures/03_校正前后三法验证.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图3 已生成，校正后 γ=%.1f°' % pm1)
```

### 运行与结果

![校正前后三法验证](./figures/03_校正前后三法验证.png)

*图3 左：校正后阶跃响应超调大幅下降、调节时间缩短；中：根轨迹极点左移（更稳）；右：Bode 幅频抬升、ωc 右移、γ 从 30° 提到 44°（接近 45° 目标）。三法结论一致：达标。*

**现象解释**：这步是"设计→验证"的收尾——
- **时域**：σ% 从 50% 降到约 30%、ts 缩短 → 快、稳都改善；
- **根轨迹**：校正装置加了零点，根轨迹被"掰"向左，阻尼变大；
- **频域**：γ 从 30° 提升到 44°（目标 45°，差 1° 属于设计余量不足，工程上可加大 φm 再迭代一次，或用仿真自动搜索）。
**三个独立视角给出同一个结论 → 设计可靠。**

## 图4 全流程流水线

### 绘图代码

```python
"""第9章图4：从建模到验证的七步流水线"""
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(12, 3.6))
ax.axis('off')
steps = [
    ('1. 物理建模', '电路+转矩方程\n→ G(s)', '#263238'),
    ('2. 时域分析', 'Routh 判稳\ness、σ%、ts', '#1565C0'),
    ('3. 根轨迹', '极点走位\n临界 K、ζ', '#C62828'),
    ('4. 频域分析', 'Bode/Nyquist\nγ、ωc', '#2E7D32'),
    ('5. 发现问题', 'γ=28°<45°\n不达标', '#E65100'),
    ('6. 校正设计', '超前/滞后/PID\n算 a、T', '#6A1B9A'),
    ('7. 仿真验证', '三法核对\n达标？', '#00838F'),
]
x = np.linspace(0.02, 0.98, len(steps))
for xi, (title, sub, col) in zip(x, steps):
    box = dict(boxstyle='round,pad=0.45', facecolor=col, alpha=0.85)
    ax.text(xi, 0.6, title, ha='center', va='center', fontsize=11,
            color='white', fontweight='bold', bbox=box)
    ax.text(xi, 0.1, sub, ha='center', va='center', fontsize=9, color='#333')
for i in range(len(x)-1):
    ax.annotate('', xy=(x[i+1]-0.012, 0.6), xytext=(x[i]+0.012, 0.6),
                arrowprops=dict(arrowstyle='->', lw=2, color='#333'))
ax.text(0.5, 0.93, '任何系统的"从零到达标"流水线', ha='center', fontsize=13, color='#1A237E', fontweight='bold')
ax.text(0.5, -0.05, '若第7步不达标 → 回到第6步迭代（工程常态）', ha='center', fontsize=10, color='#C62828')
fig.tight_layout()
fig.savefig('figures/04_全流程流水线.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图4 已生成')
```

### 运行与结果

![全流程流水线](./figures/04_全流程流水线.png)

*图4 从建模到验证的七步流水线。每一步对应本书的一章：建模(3)→时域(5)→根轨迹(6)→频域(7)→校正(8)。不达标就回到第6步迭代。*

**现象解释**：这张图是"工程师思维"的浓缩——任何系统（电机、温控、无人机姿态、化工过程）都可以套这条流水线。**它不是自动控制原理的"考点"，而是你未来工作的方法论。**
