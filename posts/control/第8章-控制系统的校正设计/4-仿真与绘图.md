> 本章目标：直观对比**超前/滞后校正的频率特性**、看清 **PID 三参数各自的作用**、演示**校正前后阶跃响应与裕度改善**，并记住**校正设计六步流程**。

## 图1 超前与滞后校正的频率特性对比

### 绘图代码

```python
"""第8章图1：超前/滞后校正的 Bode 特性"""
import numpy as np
import matplotlib.pyplot as plt
import control

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

omega = np.logspace(-2, 2, 600)
s = control.TransferFunction.s

# 超前校正 a=6，设计 ωm=5 → T = 1/(5√6)
a_lead, T_lead = 6.0, 1/(5*np.sqrt(6))
Gc_lead = (1+a_lead*T_lead*s)/(1+T_lead*s)
mag_lead, phase_lead, _ = control.frequency_response(Gc_lead, omega)

# 滞后校正 b=0.2，转折频率 0.2 与 1.0
Gc_lag = (1+0.2*5*s)/(1+5*s)
mag_lag, phase_lag, _ = control.frequency_response(Gc_lag, omega)

fig, (ax0, ax1) = plt.subplots(2, 1, figsize=(10, 7.5))
ax0.semilogx(omega, 20*np.log10(np.maximum(mag_lead,1e-6)), lw=2.4, color='#1565C0', label='超前校正（+10lg a 提幅）')
ax0.semilogx(omega, 20*np.log10(np.maximum(mag_lag,1e-6)), lw=2.4, color='#C62828', label='滞后校正（高频衰减）')
ax0.axhline(0, color='k', lw=0.8)
ax0.set_ylabel('幅值 (dB)'); ax0.grid(alpha=0.3, which='both'); ax0.legend(fontsize=10, loc='center left')
ax0.set_title('超前与滞后校正的频率特性：一个加相位，一个压幅值', fontsize=12)

ax1.semilogx(omega, np.degrees(phase_lead), lw=2.4, color='#1565C0', label='超前校正（相位 +）')
ax1.semilogx(omega, np.degrees(phase_lag), lw=2.4, color='#C62828', label='滞后校正（相位 -）')
ax1.axhline(0, color='k', lw=0.8)
ax1.set_xlabel('ω (rad/s)'); ax1.set_ylabel('相角 (度)'); ax1.grid(alpha=0.3, which='both')
ax1.legend(fontsize=10, loc='lower left')
fig.tight_layout()
fig.savefig('figures/01_超前与滞后校正Bode对比.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图1 已生成')
```

### 运行与结果

![超前与滞后校正Bode对比](./figures/01_超前与滞后校正Bode对比.png)

*图1 左（幅频）：超前校正把中频段幅值抬高 +10lg a dB（带来相位超前）；滞后校正高频段衰减 20lg b dB（压幅值）。右（相频）：超前校正相位为正（0°→+φm→0°），滞后校正相位为负。*

**现象解释**：两张 Bode 图一对比，两句话就刻在脑子里：
- **超前 = 加相位**：相角抬成正的，同时幅值抬高——"提神"；
- **滞后 = 减相位**：相角是负的，但高频幅值被压低——"压担子"。
滞后校正靠"压幅值让 ωc 左移"来间接提 γ，而不是靠加相位——这正是两类校正最本质的区别。

## 图2 PID 三个参数的作用

### 绘图代码

```python
"""第8章图2：对象 G=1/(s²+s) 分别配 P / PI / PD / PID 的阶跃响应"""
import numpy as np
import matplotlib.pyplot as plt
import control

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

s = control.TransferFunction.s
Gp = 1/(s**2 + s)

def fb(cont):
    return control.feedback(cont*Gp, 1)

t = np.linspace(0, 30, 1200)
C_P   = fb(2.0)                  # 纯 P：快但有稳态误差
C_PI  = fb(2.0 + 0.8/s)          # PI：稳态误差归零，但更振
C_PD  = fb(2.0 + 1.2*s)          # PD：超调明显减小
C_PID = fb(2.0 + 0.8/s + 1.2*s)  # PID：取长补短

configs = [('纯 P：响应快但有稳态误差', C_P, '#607D8B'),
           ('PI：稳态误差归零，振荡增大', C_PI, '#C62828'),
           ('PD：超调明显减小', C_PD, '#1565C0'),
           ('PID：三者取长补短', C_PID, '#2E7D32')]

fig, ax = plt.subplots(figsize=(10, 6))
for lab, sys, col in configs:
    T, y = control.step_response(sys, t)
    ax.plot(T, y, lw=2.2, color=col, label=lab)
ax.axhline(1, color='k', ls=':', lw=1)
ax.set_xlim(0, 30); ax.set_ylim(-0.4, 2.0)
ax.set_xlabel('t (s)'); ax.set_ylabel('c(t)')
ax.set_title('PID 三个参数的作用：P 快、I 准、D 稳', fontsize=12)
ax.legend(fontsize=10); ax.grid(alpha=0.3)
fig.tight_layout()
fig.savefig('figures/02_PID三个参数的作用.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图2 已生成')
```

### 运行与结果

![PID三个参数的作用](./figures/02_PID三个参数的作用.png)

*图2 同一对象 G=1/(s²+s)：纯 P（灰）响应快但稳态误差约 0.33；PI（红）稳态误差归零但振荡明显增大；PD（蓝）超调小；PID（绿）超调小、稳态误差为零、响应也快——三者取长补短。*

**现象解释**：**"P 快、I 准、D 稳"六个字全在这张图上**：
- P 不加 I：终值停在 2/3≈0.67（有静差）；
- 加了 I（红）：终值回到 1（静差消除），但超调变大——积分是"翻旧账"，反应慢还容易过冲；
- 加了 D（蓝）：超调从 0.6 压到 0.05 左右——微分"看趋势提前刹车"；
- 三者全上（绿）：又快、又准、又稳。

## 图3 校正前后阶跃响应对比（超前校正完整案例）

### 绘图代码

```python
"""第8章图3：对 G=10/[s(s+1)(s+5)]（γ=25.4° 不达标）做超前校正"""
import numpy as np
import matplotlib.pyplot as plt
import control

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

s = control.TransferFunction.s
G0 = 10/(s*(s+1)*(s+5))

# 设计：φm = 45°-25.4°+12° ≈ 32°，a = (1+sinφm)/(1-sinφm) ≈ 3.2
phi_m = np.radians(31.6)
a = (1+np.sin(phi_m))/(1-np.sin(phi_m))
omega_s = np.logspace(-1.5, 0.5, 4000)
mag0, _, _ = control.frequency_response(G0, omega_s)
idx = np.argmin(np.abs(20*np.log10(np.maximum(mag0,1e-9)) + 10*np.log10(a)))
wm = omega_s[idx]
T_lead = 1/(wm*np.sqrt(a))
Gc = (1+a*T_lead*s)/(1+T_lead*s)      # 超前校正装置

sys0 = control.feedback(G0, 1)
sys1 = control.feedback(Gc*G0, 1)
gm0, pm0, _, _ = control.margin(G0)
gm1, pm1, _, _ = control.margin(Gc*G0)

t = np.linspace(0, 20, 1200)
fig, (ax0, ax1) = plt.subplots(1, 2, figsize=(13, 5.5))
T, y0 = control.step_response(sys0, t)
T, y1 = control.step_response(sys1, t)
ax0.plot(T, y0, lw=2.2, color='#C62828', label=f'校正前：γ={pm0:.1f}°')
ax0.plot(T, y1, lw=2.2, color='#2E7D32', label=f'校正后：γ={pm1:.1f}°')
ax0.axhline(1, color='k', ls=':', lw=1)
ax0.set_xlim(0, 20); ax0.set_ylim(-0.3, 1.6)
ax0.set_xlabel('t (s)'); ax0.set_ylabel('c(t)')
ax0.set_title('阶跃响应：超前校正前后', fontsize=12)
ax0.legend(fontsize=10); ax0.grid(alpha=0.3)

omega_b = np.logspace(-2, 2, 600)
mag0b, _, _ = control.frequency_response(G0, omega_b)
mag1b, _, _ = control.frequency_response(Gc*G0, omega_b)
ax1.semilogx(omega_b, 20*np.log10(np.maximum(mag0b,1e-9)), lw=2.2, color='#C62828', label='校正前幅频')
ax1.semilogx(omega_b, 20*np.log10(np.maximum(mag1b,1e-9)), lw=2.2, color='#2E7D32', label='校正后幅频')
ax1.axhline(0, color='k', lw=0.8)
ax1.set_xlabel('ω (rad/s)'); ax1.set_ylabel('幅值 (dB)')
ax1.set_title('Bode 幅频对比：校正后 ωc 右移', fontsize=12)
ax1.legend(fontsize=10, loc='lower left'); ax1.grid(alpha=0.3, which='both')
fig.tight_layout()
fig.savefig('figures/03_校正前后阶跃响应对比.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图3 已生成，校正前 γ=%.1f°，校正后 γ=%.1f°' % (pm0, pm1))
```

### 运行与结果

![校正前后阶跃响应对比](./figures/03_校正前后阶跃响应对比.png)

*图3 左：校正前（红）γ=25.4°、振荡明显、超调约 50%；校正后（绿）γ≈43°、超调约 30%、调节时间缩短。右：Bode 幅频对比，校正后的幅频曲线在中频段被抬高、ωc 从 1.23 右移到更高频率。*

**现象解释**：这就是"设计→验证"的完整闭环：
- 校正前系统**稳定但太脆**（γ=25.4°<45°），响应超调大、振荡久；
- 设计超前校正（φm=31.6°，a≈3.2），把 ωc 右移、γ 提到 ≈43°（接近 45° 目标）；
- 效果：超调明显下降、响应更快——**"稳"和"快"同时改善**。
- 注意：验证结果 43° 与目标 45° 还有小差距（ε 补偿角取 12° 已尽量贴近）。**工程上这是常态**——设计值不达标就再迭代一次（加大 φm 或微调），直到达标。

## 图4 校正设计全流程

### 绘图代码

```python
"""第8章图4：校正设计六步流程示意"""
import numpy as np
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(11, 4))
ax.axis('off')
steps = [
    ('1. 定指标', 'γ≥45°\nσ%≤20%\ness 达标', '#1565C0'),
    ('2. 定增益', '按 K_v/稳态\n误差求 K', '#E65100'),
    ('3. 找差距', '量 γ0、ωc0\n对比目标', '#C62828'),
    ('4. 选打法', '超前/滞后/PID\n反馈/复合', '#6A1B9A'),
    ('5. 算参数', 'a、b、T\n或 Kp Ki Kd', '#00838F'),
    ('6. 验证', 'Bode 图+阶跃\n响应达标', '#2E7D32'),
]
x = np.linspace(0.02, 0.98, len(steps))
for xi, (title, sub, col) in zip(x, steps):
    box = dict(boxstyle='round,pad=0.5', facecolor=col, alpha=0.85)
    ax.text(xi, 0.55, title, ha='center', va='center', fontsize=13,
            color='white', fontweight='bold', bbox=box)
    ax.text(xi, 0.12, sub, ha='center', va='center', fontsize=10, color='#333')
for i in range(len(x)-1):
    ax.annotate('', xy=(x[i+1]-0.015, 0.55), xytext=(x[i]+0.015, 0.55),
                arrowprops=dict(arrowstyle='->', lw=2, color='#333'))
ax.set_title('校正设计六步流程：从指标到验证（本图是全章的"施工图"）', fontsize=13)
fig.tight_layout()
fig.savefig('figures/04_校正设计全流程.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图4 已生成')
```

### 运行与结果

![校正设计全流程](./figures/04_校正设计全流程.png)

*图4 校正设计六步：定指标 → 定增益 → 找差距 → 选打法 → 算参数 → 验证。这是一张"施工图"，做任何校正题都按这个顺序走。*

**现象解释**：图3 的完整设计就是按图4 的六步执行的。做题或做工程时，把这张图贴在旁边，按顺序一步步来，**每一步都有明确的计算方法**（见公式推导.md），就不会乱。特别提醒：第6步"验证"不是可选项——设计算完必须回验裕度和响应，不达标就回到第3步迭代。
