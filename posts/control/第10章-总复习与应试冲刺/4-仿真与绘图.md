"""第10章图1：全课知识地图"""

    ('时域法\nRouth+σ%,ts', (0.18, 0.42), '#1565C0'),

        fontsize=9.5, color='#555', style='italic')

print('图1 已生成')

## 图2 稳准快总纲

### 绘图代码

```python
"""第10章图2：稳准快总纲"""
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(11, 5.5))
ax.axis('off')

ax.text(0.5, 0.94, '全书总纲：稳、准、快', ha='center', fontsize=16,
        color='#1A237E', fontweight='bold')

items = [
    ('稳', '系统会不会失稳？\nRouth / 根轨迹 / γ>0 / Nyquist', '#C62828', 0.2, 0.6),
    ('准', '误差多大？\n型别、K_p/K_v/K_a、ess 表', '#1565C0', 0.5, 0.6),
    ('快', '多快到达？\nσ%、ts、tp；ζ、ωn；γ、ωc', '#2E7D32', 0.8, 0.6),
]
for ch, sub, col, x, y in items:
    box = dict(boxstyle='round,pad=0.6', facecolor=col, alpha=0.9)
    ax.text(x, y, ch, ha='center', va='center', fontsize=26, color='white', bbox=box, fontweight='bold')
    ax.text(x, y-0.16, sub, ha='center', va='center', fontsize=10.5, color='#333')

methods = [
    ('时域法', 'Routh 表\nσ%、ts、ess', '#1565C0', 0.17, 0.18),
    ('根轨迹法', '极点走位\nζ、ωn、K范围', '#C62828', 0.5, 0.18),
    ('频域法', 'Bode/Nyquist\nγ、Kg', '#2E7D32', 0.83, 0.18),
]
for name, sub, col, x, y in methods:
    box = dict(boxstyle='round,pad=0.45', facecolor=col, alpha=0.85)
    ax.text(x, y, f'{name}\n{sub}', ha='center', va='center', fontsize=10.5,
            color='white', bbox=box)
ax.text(0.5, 0.42, '▲ 三大方法从三个角度回答，经 ζ 互相换算 ▲', ha='center', fontsize=10,
        color='#555')
ax.text(0.5, 0.02, '答题先问：这题考哪个字？→ 选武器 → 套流程', ha='center',
        fontsize=11, color='#E65100', fontweight='bold')
fig.tight_layout()
fig.savefig('figures/02_稳准快总纲.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图2 已生成')
```

### 运行与结果

![稳准快总纲](./figures/02_稳准快总纲.png)

*图2 三个大字：稳、准、快，分别对应"会不会失稳""误差多大""多快到达"。下面三个方法框回答同一套问题。*

**现象解释**：这是答题的第一反应训练——**看到题目先归类**：问稳定范围 → "稳"，用 Routh/根轨迹；问稳态误差 → "准"，查型别表；问超调/调节时间 → "快"，套二阶公式或换算 γ。

## 图3 三大方法对比思维导图

### 绘图代码

```python
"""第10章图3：三大方法对比思维导图"""
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(12, 7))
ax.axis('off')

ax.text(0.5, 0.95, '三大分析方法横向对比', ha='center', fontsize=15,
        color='#1A237E', fontweight='bold')

cols = [('#1565C0', '时域法'), ('#C62828', '根轨迹法'), ('#2E7D32', '频域法')]
rows = ['看什么', '判稳', '快(动态)', '准(稳态)', '优势', '局限']
cells = [
    ('响应 c(t)', '极点走位', '频率特性 G(jw)'),
    ('Routh 表', '不过虚轴', 'γ>0 或 Nyquist'),
    ('σ%、ts 直接算', 'ζ、ωn 极点位置', 'γ<=>ζ、ωc<=>ts'),
    ('误差系数/型别', '增益 K 决定型别', '低频增益/斜率'),
    ('直观、物理意义清', '看清参数变化影响', '实验建模·抗噪'),
    ('需固定参数', '绘制繁琐', '相对抽象'),
]

for i, (col, name) in enumerate(cols):
    x = 0.18 + i*0.27
    box = dict(boxstyle='round,pad=0.4', facecolor=col, alpha=0.9)
    ax.text(x, 0.86, name, ha='center', va='center', fontsize=12,
            color='white', bbox=box, fontweight='bold')

for r, row in enumerate(rows):
    y = 0.78 - r*0.115
    ax.text(0.05, y, row, ha='center', va='center', fontsize=10.5, color='#333', fontweight='bold')
    for i, cell in enumerate(cells[r]):
        x = 0.18 + i*0.27
        ax.text(x, y, cell, ha='center', va='center', fontsize=9.5, color='#333')

ax.text(0.5, 0.08, '三副眼镜 · 同一结论 · 会师点 ζ（γ≈100ζ <=> σ% <=> 极点夹角）', ha='center',
        fontsize=11, color='#E65100', fontweight='bold')
fig.tight_layout()
fig.savefig('figures/03_三大方法对比思维导图.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图3 已生成')
```

### 运行与结果

![三大方法对比思维导图](./figures/03_三大方法对比思维导图.png)

*图3 三大方法在六个维度（看什么/判稳/快/准/优势/局限）上的对比表。考试常考"用三种方法分析同一系统"，此表就是答题框架。*

**现象解释**：这张表是第9章对比表的可视化版。**关键看最后一行"局限"**——知道每种方法的短板，才能在不同场景选对武器（例如：只有实验数据时只能上频域法）。

## 图4 考前冲刺路线图

### 绘图代码

```python
"""第10章图4：考前冲刺路线图"""
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

fig, ax = plt.subplots(figsize=(11, 4.5))
ax.axis('off')

ax.text(0.5, 0.9, '考前冲刺：三轮复习法', ha='center', fontsize=15,
        color='#1A237E', fontweight='bold')

rounds = [
    ('第一轮 串网\n考前7~5天\n每天2章过公式\n·知识地图定位\n·公式能默写', '#1565C0'),
    ('第二轮 刷题\n考前4~2天\n各章提高题+综合卷\n·先定位再选武器\n·错题回看易错表', '#C62828'),
    ('第三轮 模拟\n考前1天\n限时1套综合卷\n·五大题型全过\n·睡前看易错点', '#2E7D32'),
]
for i, (txt, col) in enumerate(rounds):
    x = 0.16 + i*0.34
    box = dict(boxstyle='round,pad=0.55', facecolor=col, alpha=0.88)
    ax.text(x, 0.42, txt, ha='center', va='center', fontsize=10.5, color='white', bbox=box)
    if i < 2:
        ax.annotate('', xy=(x+0.16, 0.42), xytext=(x+0.14, 0.42),
                    arrowprops=dict(arrowstyle='->', lw=2.4, color='#555'))

ax.text(0.5, 0.12, '考场时间建议：建模15min · 时域15min · 根轨迹20min · Bode20min · 校正20min · 检查10min', ha='center',
        fontsize=10, color='#555')
fig.tight_layout()
fig.savefig('figures/04_考前冲刺路线图.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图4 已生成')
```

### 运行与结果

![考前冲刺路线图](./figures/04_考前冲刺路线图.png)

*图4 三轮复习法：串网（7~5天）→ 刷题（4~2天）→ 模拟（1天），附考场时间分配建议。*

**现象解释**：复习不是"重新学一遍"，而是"由宽到窄"——先串知识网（全覆盖），再刷题找漏洞（收缩范围），最后模拟练手感（实战状态）。
