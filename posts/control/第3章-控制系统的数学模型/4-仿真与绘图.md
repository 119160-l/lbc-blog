## 图1 建模链示意图

### 绘图代码

```python
"""第3章图1：建模链条"""
import matplotlib.pyplot as plt
from matplotlib.patches import FancyBboxPatch

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

def box(ax, x, y, w, h, text, fc='#E3F2FD', ec='#1565C0', fs=10):
    p = FancyBboxPatch((x, y), w, h, boxstyle="round,pad=0.02", fc=fc, ec=ec, lw=1.6)
    ax.add_patch(p)
    ax.text(x + w/2, y + h/2, text, ha='center', va='center', fontsize=fs)

def arrow(ax, x1, y1, x2, y2, text=None, ty=0.03, fs=9):
    ax.annotate('', xy=(x2, y2), xytext=(x1, y1),
                arrowprops=dict(arrowstyle='-|>', color='#333333', lw=1.6))
    if text:
        ax.text((x1+x2)/2, (y1+y2)/2 + ty, text, ha='center', va='center', fontsize=fs)

fig, ax = plt.subplots(figsize=(11, 2.6))
ax.set_xlim(0, 12); ax.set_ylim(0, 3); ax.axis('off')
ax.set_title('建模链条：物理世界 → 数学世界', fontsize=13)
box(ax, 0.2, 1.1, 1.9, 1.0, '物理系统\n(电路/电机)', fc='#FFEBEE', ec='#C62828')
box(ax, 3.1, 1.1, 1.9, 1.0, '物理规律\n(欧姆/牛顿定律)', fc='#FFF3E0', ec='#E65100')
box(ax, 6.0, 1.1, 1.9, 1.0, '微分方程', fc='#FFF9C4', ec='#F9A825')
box(ax, 8.9, 1.1, 1.9, 1.0, '传递函数\nG(s)', fc='#E8F5E9', ec='#2E7D32')
arrow(ax, 2.1, 1.6, 3.1, 1.6, '① 抽象简化', 0.1)
arrow(ax, 5.0, 1.6, 6.0, 1.6, '② 列方程', 0.1)
arrow(ax, 7.9, 1.6, 8.9, 1.6, '③ 拉氏变换', 0.1)
ax.text(6.0, 0.4, '④ 用 G(s) 做分析：乘除、画图、判稳、设计（第4~8章）', fontsize=10, ha='center', color='#2E7D32')
fig.tight_layout()
fig.savefig('figures/01_建模链示意图.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图1 已生成')
```

### 运行与结果

![建模链示意图](./figures/01_建模链示意图.png)

*图1 建模四步：抽象简化 → 列微分方程 → 拉氏变换 → 得到传递函数，之后所有分析都基于 G(s)。*

**现象解释**：这张图是本课所有章节的"交通图"——第3章走到最右端的 $G(s)$，第4章在 $G(s)$ 之间做连接与化简，第5~8章全部围绕 $G(s)$ 展开分析。

## 图2 六大典型环节阶跃响应对比

### 绘图代码

```python
"""第3章图2：六大典型环节单位阶跃响应"""
import numpy as np
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

t = np.linspace(0, 6, 800)
fig, axes = plt.subplots(2, 3, figsize=(13, 6.5))

def plot_step(ax, y, title):
    ax.plot(t, y, lw=2.2, color='#1565C0')
    ax.axhline(1, color='k', ls=':', lw=1, alpha=0.5)
    ax.set_title(title, fontsize=11)
    ax.set_xlim(0, 6); ax.grid(alpha=0.3); ax.set_xlabel('t')

plot_step(axes[0,0], 2*np.ones_like(t), '比例环节 K=2：立即跟随（2倍）')
plot_step(axes[0,1], t, '积分环节 1/s：线性增长')
plot_step(axes[0,2], np.ones_like(t), '微分环节（示意）：输入为斜坡')
wn = 3; zeta = 0.3
s = np.sqrt(1-zeta**2)
plot_step(axes[1,0], 1-np.exp(-t), '惯性环节 1/(s+1)：指数趋近')
plot_step(axes[1,1], 1 - np.exp(-zeta*wn*t)*(np.cos(wn*s*t) + zeta/s*np.sin(wn*s*t)), '振荡环节：衰减振荡')
plot_step(axes[1,2], np.where(t>1, 1, 0), '延迟环节 e^(-s)：平移滞后')
for i in range(2):
    for j in range(3):
        axes[i,j].set_ylim(-0.5, 2.2)
fig.suptitle('六大典型环节的单位阶跃响应形态', fontsize=14)
fig.tight_layout()
fig.savefig('figures/02_典型环节阶跃响应对比.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图2 已生成')
```

### 运行与结果

![典型环节阶跃响应对比](./figures/02_典型环节阶跃响应对比.png)

*图2 六大环节的响应形态：比例=直跟、积分=爬升、微分=示踪、惯性=指数趋近、振荡=衰减摆动、延迟=平移滞后。*

**现象解释**：一眼就能识别"长什么样的是哪个环节"——这是后续"看响应猜模型"技能的训练。特别注意**振荡环节**的形态，第5章会把它拆开研究透。

## 图3 RC 电路：时间常数对响应的影响

### 绘图代码

```python
"""第3章图3：不同时间常数的RC响应"""
import numpy as np
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

t = np.linspace(0, 6, 800)
fig, ax = plt.subplots(figsize=(9, 5))
for T, c in [(0.5, '#1565C0'), (1.0, '#2E7D32'), (2.0, '#E65100'), (4.0, '#C62828')]:
    ax.plot(t, 1 - np.exp(-t/T), lw=2.2, color=c, label=f'T = RC = {T} s')
ax.axhline(1, color='k', ls=':', lw=1, alpha=0.6)
ax.set_xlabel('t (s)'); ax.set_ylabel('u_o(t)')
ax.set_title('RC 电路（惯性环节）单位阶跃响应：T 越大响应越慢', fontsize=13)
ax.legend(fontsize=11); ax.grid(alpha=0.3)
ax.set_ylim(0, 1.15)
fig.tight_layout()
fig.savefig('figures/03_RC电路不同时间常数响应.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图3 已生成')
```

### 运行与结果

![RC电路不同时间常数响应](./figures/03_RC电路不同时间常数响应.png)

*图3 四条曲线分别是 $T=0.5,1,2,4$ 秒的阶跃响应。$T$ 越小上升越快。*

**现象解释**：
- $t=T$ 时响应到达稳态的 63.2%（$1-e^{-1}\approx0.632$）——**时间常数的直观定义**；
- $t=3T$ 时到达 95%，$t=5T$ 时到达 99.3%——**工程上常说"3T~5T 系统基本稳定"**；
- 这就是为什么"惯性大（T大）的系统反应慢"，比如大房间的空调比小房间慢得多。

## 图4 非线性系统的线性化

### 绘图代码

```python
"""第3章图4：小偏差线性化示意图（y=x² 在 x0=2）"""
import numpy as np
import matplotlib.pyplot as plt

plt.rcParams['font.sans-serif'] = ['PingFang SC', 'STHeiti', 'SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

x = np.linspace(0.5, 3.5, 300)
fig, ax = plt.subplots(figsize=(8.5, 6))
ax.plot(x, x**2, lw=2.4, color='#1565C0', label='非线性曲线 y = x^2')
xt = np.linspace(1.2, 2.8, 100)
ax.plot(xt, 4 + 4*(xt-2), '--', lw=2.2, color='#C62828', label='线性化切线：Δy ≈ 4Δx（在 x_0=2）')
ax.plot([2], [4], 'o', ms=9, color='#C62828', zorder=5)
ax.annotate('工作点 (2, 4)', xy=(2, 4), xytext=(2.25, 4.6), fontsize=11, color='#C62828')
ax.annotate('小偏差区间：直线≈曲线', xy=(1.65, 3.2), xytext=(0.7, 6.8), fontsize=10,
            arrowprops=dict(arrowstyle='->', color='#333'), color='#333')
ax.annotate('大偏差：直线偏离曲线', xy=(3.0, 9), xytext=(2.4, 8.3), fontsize=10,
            arrowprops=dict(arrowstyle='->', color='#333'), color='#333')
ax.set_xlabel('x'); ax.set_ylabel('y')
ax.set_title('小偏差线性化：切线只在小范围内有效', fontsize=13)
ax.legend(fontsize=11); ax.grid(alpha=0.3)
fig.tight_layout()
fig.savefig('figures/04_非线性线性化示意图.png', dpi=150, bbox_inches='tight', facecolor='white')
plt.close(fig)
print('图4 已生成')
```

### 运行与结果

![非线性线性化示意图](./figures/04_非线性线性化示意图.png)

*图4 曲线 $y=x^2$ 在 $(2,4)$ 处的切线。切点附近直线与曲线贴合，远离切点则严重偏离。*

**现象解释**：线性化=用切线代替曲线。切点附近（小偏差）误差可以忽略；离开切点越远误差越大。**这正是"小偏差线性化"六个字的含义**——它也解释了为什么线性系统分析适合"扰动抑制"场景（扰动通常是小偏差），而不适合"大范围指令跟踪"。
